\begin{theorem}

	Let $A \subseteq \ ^{*}\mathbb{R}$ definable with parameters, bounded. Then in $ \ ^{*}\mathbb{R}$ there exists $sup(A)$.
  
\end{theorem}

\begin{proof}

	Let $A = \phi(^{*}\mathbb{R},t_1,..,t_n)$ where $t_1,..,t_n \in \ ^{*}\mathbb{R}$.
	For simplicitity we may assume only one parameter is involved.
	Since
	\begin{align*}
		\mathbb{R} \models \forall y \bigg( & \big( \exists c \forall x (\phi(x,y) \to x \leq c) \big) \to \\
						& \big( \exists b \big( \forall x \phi(x,y) \to y \leq b \big) 
								\land \big( \forall z (\forall x \phi(x,y) \to x \leq z) \to z \geq b \big) \big) \bigg)
	\end{align*}
	the same holds in $^{*} \mathbb{R}$.
\end{proof}
